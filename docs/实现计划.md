```python
import sqlite3
from pathlib import Path

SCHEMA_PATH = Path(__file__).with_name("schema.sql")

def connect(db_path: str) -> sqlite3.Connection:
    Path(db_path).parent.mkdir(parents=True, exist_ok=True)
    conn = sqlite3.connect(db_path)
    conn.row_factory = sqlite3.Row
    conn.execute("PRAGMA foreign_keys = ON")
    return conn

def init_db(conn) -> None:
    conn.executescript(SCHEMA_PATH.read_text(encoding="utf-8"))
    conn.commit()

def upsert(conn, table: str, row: dict, keys: list[str]) -> None:
    cols = list(row)
    updates = ",".join(f"{c}=excluded.{c}" for c in cols if c not in keys)
    sql = (f"INSERT INTO {table} ({','.join(cols)}) VALUES ({','.join('?' * len(cols))}) "
           f"ON CONFLICT({','.join(keys)}) DO UPDATE SET {updates}")
    conn.execute(sql, [row[c] for c in cols])
    conn.commit()
```

```python
import csv, io, time
from datetime import datetime
import httpx

BASE = "https://www.football-data.co.uk/mmz4281/{season}/{div}.csv"
ODDS_COLS = {
    "avg":  {"h": "AvgH",  "d": "AvgD",  "a": "AvgA"},
    "max":  {"h": "MaxH",  "d": "MaxD",  "a": "MaxA"},
    "b365": {"h": "B365H", "d": "B365D", "a": "B365A"},
}

def parse_date(s: str) -> str:
    for fmt in ("%d/%m/%Y", "%d/%m/%y"):
        try:
            return datetime.strptime(s.strip(), fmt).date().isoformat()
        except ValueError:
            continue
    raise ValueError(f"无法解析日期: {s}")

def download(season: str, div: str, timeout: int = 30) -> str:
    url = BASE.format(season=season, div=div)
    last = None
    for attempt in range(3):
        try:
            r = httpx.get(url, timeout=timeout, follow_redirects=True)
            r.raise_for_status()
            return r.content.decode("utf-8-sig")
        except Exception as e:      # 网络抖动重试
            last = e
            time.sleep(2 ** attempt)
    raise RuntimeError(f"下载失败 {url}") from last

def parse_csv(text: str) -> list[dict]:
    out = []
    for row in csv.DictReader(io.StringIO(text)):
        if not (row.get("HomeTeam") or "").strip():
            continue
        odds = {}
        for name, cols in ODDS_COLS.items():
            try:
                odds[name] = {k: float(row[c]) for k, c in cols.items()}
            except (KeyError, TypeError, ValueError):
                pass
        shots = {}
        for key, col in (("hs", "HS"), ("as", "AS"), ("hst", "HST"), ("ast", "AST")):
            try:
                shots[key] = int(row[col])
            except (KeyError, TypeError, ValueError):
                pass
        out.append({
            "date": parse_date(row["Date"]),
            "home": row["HomeTeam"].strip(), "away": row["AwayTeam"].strip(),
            "home_goals": int(row["FTHG"]) if row.get("FTHG") else None,
            "away_goals": int(row["FTAG"]) if row.get("FTAG") else None,
            "result": (row.get("FTR") or "").strip() or None,
            "odds": odds, "shots": shots,
        })
    return out
```

- [ ] **Step 3: 实现**

要点：
- `SEED: dict[str, str]`（中文名 → football-data 英文名）：初始放 60+ 支常见球队（英超/德甲/意甲/西甲/法甲/英冠），如 `{"曼城": "Man City", "曼联": "Man United", "阿森纳": "Arsenal", "利物浦": "Liverpool", "切尔西": "Chelsea", ...}`（实现时按足彩常见对阵补全）；
- `normalize(name)`：去空格、全角转半角、去"足球俱乐部/FC"等后缀；
- `match(name_cn, seed=SEED)`：先精确（含别名表），再 `difflib.SequenceMatcher` 取最高相似度，≥0.86 视为模糊命中（confidence=相似度），否则 `(None, 0.0)`；
- `resolve(conn, name_cn)`：先查 `team_alias` 表（confirmed=1），再 `match`；
- `confirm(conn, name_cn, name_en)`：写 `team_alias`（confirmed=1）。

```python
import numpy as np
from scipy.optimize import brentq

def proportional(odds: list[float]) -> list[float]:
    p = 1.0 / np.asarray(odds, dtype=float)
    return (p / p.sum()).tolist()

def shin(odds: list[float]) -> list[float]:
    """Shin 去水：解 z 使 Σπ_i = 1。"""
    p = 1.0 / np.asarray(odds, dtype=float)
    B = p.sum()
    if B < 1.0:
        raise ValueError(f"非法赔率（隐含概率和 {B:.4f} < 1）")
    def pi(z):
        return (np.sqrt(z * z + 4 * (1 - z) * p * p / B) - z) / (2 * (1 - z))
    f = lambda z: pi(z).sum() - 1.0
    z = brentq(f, 1e-9, 0.5)
    out = pi(z)
    return (out / out.sum()).tolist()

def devig(odds_json: dict, method: str = "shin") -> list[float] | None:
    """从 matches.odds_json 取一组赔率并去水；优先 avg，退化 max/b365。"""
    if not odds_json:
        return None
    for key in ("avg", "max", "b365"):
        o = odds_json.get(key)
        if o:
            try:
                return shin([o["h"], o["d"], o["a"]]) if method == "shin" \
                    else proportional([o["h"], o["d"], o["a"]])
            except ValueError:
                continue
    return None
```

```python
import math
IDX = {"H": 0, "D": 1, "A": 2}

def logloss(probs, outcomes) -> float:
    eps = 1e-15
    return -sum(math.log(max(p[IDX[o]], eps)) for p, o in zip(probs, outcomes)) / len(outcomes)

def brier(probs, outcomes) -> float:
    total = 0.0
    for p, o in zip(probs, outcomes):
        y = [1.0 if i == IDX[o] else 0.0 for i in range(3)]
        total += sum((pi - yi) ** 2 for pi, yi in zip(p, y))
    return total / len(outcomes)
```

```python
from dataclasses import dataclass, field
from datetime import timedelta

@dataclass
class PredRecord:
    match_id: int; date: str; probs: list; outcome: str; source: str

def walk_forward(matches, model_factory, start_date, refit_days=7, source="model"):
    """
    matches: 按日期升序的 dict 列表（id/date/home_id/away_id/result/...）
    model_factory(past_matches) -> callable(match) -> [ph, pd, pa]
    每 refit_days 重新拟合一次；始终只用 start_date 之前 + 已流过的数据。
    """
    records = []
    model, last_fit = None, None
    for i, m in enumerate(matches):
        past = matches[:i]
        if not past:
            continue
        if model is None or (m["date"] - last_fit) >= timedelta(days=refit_days):
            model = model_factory(past)
            last_fit = m["date"]
        if m["date"] < start_date or m["result"] is None:
            continue
        records.append(PredRecord(m["id"], str(m["date"]), model(m), m["result"], source))
    return records
```

```python
"""Dixon-Coles：带低比分修正与时间衰减的双泊松模型，按联赛单独拟合。"""
from datetime import date
import numpy as np
from scipy.optimize import minimize
from scipy.stats import poisson

class DixonColes:
    def __init__(self, half_life_days: float = 180.0, max_goals: int = 10):
        self.xi = 0.5 ** (1.0 / half_life_days)   # 每日衰减系数
        self.max_goals = max_goals
        self.teams: list[str] = []
        self._idx: dict[str, int] = {}
        self.a = self.b = None
        self.gamma = 0.0
        self.rho = 0.0

    # ---------- 拟合 ----------
    def fit(self, rows, teams=None):
        """rows: [(date_str|date, home, away, hg, ag)]；teams 缺省从数据推断。"""
        self.teams = teams or sorted({r[1] for r in rows} | {r[2] for r in rows})
        self._idx = {t: i for i, t in enumerate(self.teams)}
        n = len(self.teams)
        d = np.array([np.datetime64(r[0]) for r in rows])
        hi = np.array([self._idx[r[1]] for r in rows])
        ai = np.array([self._idx[r[2]] for r in rows])
        hg = np.array([r[3] for r in rows], dtype=float)
        ag = np.array([r[4] for r in rows], dtype=float)
        w = self.xi ** (d.max() - d).astype(float)

        def nll(params):
            a = params[:n]; a = a - a.mean()
            b = params[n:2 * n]; b = b - b.mean()
            gamma, rho = params[2 * n], params[2 * n + 1]
            lam = np.exp(a[hi] - b[ai] + gamma)
            mu = np.exp(a[ai] - b[hi])
            tau = np.ones_like(lam)
            m = (hg == 0) & (ag == 0); tau[m] = 1 - lam[m] * mu[m] * rho
            m = (hg == 1) & (ag == 0); tau[m] = 1 + mu[m] * rho
            m = (hg == 0) & (ag == 1); tau[m] = 1 + lam[m] * rho
            m = (hg == 1) & (ag == 1); tau[m] = 1 - rho
            ll = (hg * np.log(lam) - lam + ag * np.log(mu) - mu
                  + np.log(np.maximum(tau, 1e-9)))
            return -float(np.sum(w * ll))

        x0 = np.zeros(2 * n + 2)
        x0[2 * n] = 0.25
        x0[2 * n + 1] = -0.05
        bounds = [(-3, 3)] * (2 * n) + [(0.0, 1.0), (-0.5, 0.5)]
        res = minimize(nll, x0, method="L-BFGS-B", bounds=bounds)
        a = res.x[:n]; self.a = a - a.mean()
        b = res.x[n:2 * n]; self.b = b - b.mean()
        self.gamma = float(res.x[2 * n])
        self.rho = float(res.x[2 * n + 1])
        return self

    # ---------- 预测 ----------
    def score_matrix(self, home: str, away: str) -> np.ndarray:
        i, j = self._idx[home], self._idx[away]
        lam = np.exp(self.a[i] - self.b[j] + self.gamma)
        mu = np.exp(self.a[j] - self.b[i])
        g = np.arange(0, self.max_goals + 1)
        P = np.outer(poisson.pmf(g, lam), poisson.pmf(g, mu))
        for (x, y, f) in ((0, 0, 1 - lam * mu * self.rho),
                          (1, 0, 1 + mu * self.rho),
                          (0, 1, 1 + lam * self.rho),
                          (1, 1, 1 - self.rho)):
            P[x, y] *= f
        return P / P.sum()

    def predict(self, home: str, away: str) -> list[float]:
        P = self.score_matrix(home, away)
        return [float(np.tril(P, -1).sum()), float(np.trace(P)), float(np.triu(P, 1).sum())]
```

```python
"""逐场按时间序构建特征：任何特征只来自此前已喂入的比赛。"""
from collections import defaultdict, deque
import numpy as np

FEATURE_COLUMNS = [
    "home_points_5", "home_points_10", "home_gf_5", "home_ga_5",
    "away_points_5", "away_points_10", "away_gf_5", "away_ga_5",
    "home_home_points_5", "away_away_points_5",
    "home_rest_days", "away_rest_days",
    "elo_home", "elo_away", "elo_diff",
    "league_avg_goals", "league_home_win_rate", "league_draw_rate",
    "season_progress",
]

class _Elo:
    K, HOME_ADV = 20.0, 65.0
    def __init__(self): self.r = defaultdict(lambda: 1500.0)
    def exp_score(self, h, a): return 1 / (1 + 10 ** ((self.r[a] - self.r[h] - self.HOME_ADV) / 400))
    def update(self, h, a, sh):
        d = self.K * (sh - self.exp_score(h, a))
        self.r[h] += d; self.r[a] -= d

class FeatureBuilder:
    WINDOW = 10
    def __init__(self):
        self.results = defaultdict(lambda: deque(maxlen=self.WINDOW))   # team -> [(pts, gf, ga, is_home)]
        self.last_date = {}
        self.elo = _Elo()
        self.league_stats = defaultdict(lambda: [0, 0, 0, 0, 0])  # goals, matches, home_wins, draws, ...
        self.season_matches = defaultdict(lambda: defaultdict(int))

    def build(self, matches):
        feats = {}
        for m in matches:                      # matches 必须按日期升序
            if m["result"]:
                feats[m["id"]] = self._row(m)
                self._update(m)
        return feats

    def build_forward(self, history, upcoming):
        """喂入全部历史后，为未赛对阵（无 result）生成特征行；当期预测路径使用。"""
        self.build(history)
        return {m["id"]: self._row(m) for m in upcoming}

    def _agg(self, team, n, home_only=None):
        rs = [r for r in self.results[team] if home_only is None or r[3] == home_only][-n:]
        if not rs: return 0.0, 0.0, 0.0
        pts = sum(r[0] for r in rs) / len(rs)
        gf = sum(r[1] for r in rs) / len(rs)
        ga = sum(r[2] for r in rs) / len(rs)
        return pts, gf, ga

    def _row(self, m):
        h, a = m["home_id"], m["away_id"]
        hp5, hgf5, hga5 = self._agg(h, 5)
        hp10, _, _ = self._agg(h, 10)
        ap5, agf5, aga5 = self._agg(a, 5)
        ap10, _, _ = self._agg(a, 10)
        hh5, _, _ = self._agg(h, 5, home_only=True)
        aa5, _, _ = self._agg(a, 5, home_only=False)
        ls = self.league_stats[m["league_code"]]
        avg_goals = (ls[0] / ls[1]) if ls[1] else 2.6
        hw_rate = (ls[2] / ls[1]) if ls[1] else 0.45
        draw_rate = (ls[3] / ls[1]) if ls[1] else 0.26
        season_total = 380  # 用历史同联赛赛季总轮次近似；实现时按联赛取实际值
        return {
            "home_points_5": hp5, "home_points_10": hp10, "home_gf_5": hgf5, "home_ga_5": hga5,
            "away_points_5": ap5, "away_points_10": ap10, "away_gf_5": agf5, "away_ga_5": aga5,
            "home_home_points_5": hh5, "away_away_points_5": aa5,
            "home_rest_days": self._rest(h, m["date"]), "away_rest_days": self._rest(a, m["date"]),
            "elo_home": self.elo.r[h], "elo_away": self.elo.r[a],
            "elo_diff": self.elo.r[h] - self.elo.r[a],
            "league_avg_goals": avg_goals, "league_home_win_rate": hw_rate, "league_draw_rate": draw_rate,
            "season_progress": min(1.0, self.season_matches[m["league_code"]][m["season"]] / season_total),
        }

    def _rest(self, team, d):
        last = self.last_date.get(team)
        return float((np.datetime64(d) - np.datetime64(last)).astype(int)) if last else 7.0

    def _update(self, m):
        sh = {"H": 1.0, "D": 0.5, "A": 0.0}[m["result"]]
        hg, ag = m["home_goals"], m["away_goals"]
        self.results[m["home_id"]].append((3 * sh, hg, ag, True))
        self.results[m["away_id"]].append((3 * (1 - sh), ag, hg, False))
        self.last_date[m["home_id"]] = m["date"]; self.last_date[m["away_id"]] = m["date"]
        self.elo.update(m["home_id"], m["away_id"], sh)
        ls = self.league_stats[m["league_code"]]
        ls[0] += hg + ag; ls[1] += 1
        ls[2] += 1 if m["result"] == "H" else 0
        ls[3] += 1 if m["result"] == "D" else 0
        self.season_matches[m["league_code"]][m["season"]] += 1
```

```python
"""覆盖概率与奖级概率（定义见设计文档 6.1）。"""
from math import prod

def coverage(probs: list[float], selections: list[str]) -> float:
    idx = {"3": 0, "1": 1, "0": 2}
    return sum(probs[idx[s]] for s in selections)

def p_first(cs: list[float]) -> float:
    return prod(cs)

def p_second(cs: list[float]) -> float:
    """对 13 场的概率（合并口径：全对时同样持有对 13 场的注）。"""
    p1 = p_first(cs)
    return p1 * sum((1 - c) / c for c in cs if c > 0)
```

```python
"""14 场与任九的方案求解。

引理（设计文档 6.2）：固定各场选择数 m_i 后，每场取概率最高的 m_i 个结果
对 P1 与 P1+P2 同时最优；故只需枚举 size 向量。
"""
import math
from itertools import product

OUTCOMES = ("3", "1", "0")

def top_m(probs: list[float], m: int) -> list[str]:
    """取概率最高的 m 个结果，按 3/1/0 顺序返回。"""
    order = sorted(range(3), key=lambda i: probs[i], reverse=True)[:m]
    return sorted((OUTCOMES[i] for i in order), key=OUTCOMES.index)

def enumerate_size_vectors(n: int, max_notes: int):
    def rec(i, prod_v, cur):
        if i == n:
            yield tuple(cur)
            return
        for m in (1, 2, 3):
            if prod_v * m <= max_notes:
                cur.append(m)
                yield from rec(i + 1, prod_v * m, cur)
                cur.pop()
    yield from rec(0, 1, [])

from football_lottery.optimizer import probabilities as P

def evaluate(probs_list, sizes):
    legs = [top_m(p, m) for p, m in zip(probs_list, sizes)]
    cs = [P.coverage(p, leg) for p, leg in zip(probs_list, legs)]
    return legs, cs, P.p_first(cs), P.p_second(cs)

def solve_14(probs_list, budget=64, ticket_price=2, objective="first_second"):
    max_notes = budget // ticket_price
    best = None
    for sizes in enumerate_size_vectors(len(probs_list), max_notes):
        legs, cs, p1, p2 = evaluate(probs_list, sizes)
        score = p1 if objective == "first" else p1 + p2
        if best is None or score > best["score"]:
            best = {"sizes": sizes, "legs": legs, "p_first": p1, "p_second": p2,
                    "p_win": p1 + p2, "score": score,
                    "notes_count": math.prod(sizes)}
    return best
```

```python
import numpy as np
NEG = -1e18

def solve_r9(probs_list, need=9, budget=64, ticket_price=2):
    """从 n 场中选 need 场并分配选择数，最大化 Πc_i。"""
    n, max_notes = len(probs_list), budget // ticket_price
    if max_notes < 1:
        raise ValueError("预算不足 2 元")
    dp = np.full((n + 1, need + 1, max_notes + 1), NEG)
    choice = {}
    dp[0, 0, 1] = 0.0
    for i in range(n):
        for k in range(need + 1):
            for notes in range(1, max_notes + 1):
                cur = dp[i, k, notes]
                if cur <= NEG / 2:
                    continue
                if k + (n - i) < need:      # 剪枝：剩余场数不够
                    continue
                if cur > dp[i + 1, k, notes]:                    # 跳过
                    dp[i + 1, k, notes] = cur
                    choice[(i + 1, k, notes)] = ("skip", None, notes)
                if k == need:
                    continue
                for m in (1, 2, 3):
                    nn = notes * m
                    if nn > max_notes:
                        continue
                    leg = top_m(probs_list[i], m)
                    c = sum(probs_list[i][OUTCOMES.index(s)] for s in leg)
                    if c <= 0:
                        continue
                    val = cur + math.log(c)
                    if val > dp[i + 1, k + 1, nn]:
                        dp[i + 1, k + 1, nn] = val
                        choice[(i + 1, k + 1, nn)] = ("pick", leg, notes)
    best_notes = max(range(1, max_notes + 1), key=lambda nt: dp[n, need, nt])
    # 回溯
    legs, k, notes = [None] * n, need, best_notes
    for i in range(n, 0, -1):
        kind, val, prev_notes = choice[(i, k, notes)]
        if kind == "pick":
            legs[i - 1] = val
            k -= 1
        else:
            legs[i - 1] = []
        notes = prev_notes
    cs = [sum(probs_list[i][OUTCOMES.index(s)] for s in leg)
          for i, leg in enumerate(legs) if leg]
    return {"legs": legs, "p_win": math.prod(cs), "p_first": math.prod(cs),
            "notes_count": best_notes}
```

```python
def hit_counts(legs, result):
    """legs: 每场选择列表；result: 每场真实结果（'3'/'1'/'0'）。
    返回 (一等奖注数, 二等奖注数)；见设计文档附录公式。"""
    n = len(legs)
    covered = [result[i] in legs[i] for i in range(n)]
    k = sum(covered)
    if k == n:
        return 1, sum(len(legs[i]) - 1 for i in range(n))
    if k == n - 1:
        miss = covered.index(False)
        return 0, len(legs[miss])
    return 0, 0

def r9_hit(legs, result):
    return 1 if all(result[i] in legs[i] for i in range(len(legs))) else 0

def amount(hit_notes: int, single_prize: float) -> float:
    return hit_notes * (single_prize or 0.0)
```

## 执行建议

- 每个任务遵循 TDD：先写测试 → 确认失败 → 最小实现 → 确认通过 → 提交（参考 @superpowers:test-driven-development）。
- 阶段之间是硬门槛：**上一阶段的验收命令没看到预期输出，不进入下一阶段**。
- 每个阶段结束把关键数字写进 `docs/回测记录.md`（运行日期、数据范围、命令、结果），它是后续所有决策的证据链。
- 遇到设计文档没覆盖的决策，先改设计文档再改代码。
